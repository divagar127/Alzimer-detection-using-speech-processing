% ===== Slide 4: Introduction and Motivation =====
\section{Introduction}
\begin{frame}{Introduction and Motivation}
    \begin{block}{Background Context}
        Alzheimer's Disease (AD) is a progressive brain disorder that gradually affects memory, thinking skills, and spoken language.
    \end{block}
    \begin{itemize}
        \item \textbf{Motivation:} Brain scans (MRI/PET) and clinical visits are expensive, time-consuming, and hard to access. Analyzing spontaneous speech provides a non-invasive, quick, and low-cost alternative for early screening.
        \item \textbf{Speech Indicators:} Subtle changes like longer pauses, unstable pitch, slower speech rate, and reduced voice clarity can indicate early cognitive decline.
        \item \textbf{Machine Learning Approach:} Extracting acoustic properties and deep features enables automated, objective classification of Alzheimer's speech.
    \end{itemize}
\end{frame}

% ===== Slide 5: Problem Definition =====
\section{Problem Definition}
\begin{frame}{Problem Definition}
    \begin{alertblock}{Main Challenges in AD Detection}
        Current diagnostic methods are costly and subjective, while traditional audio feature extraction alone yields limited accuracy.
    \end{alertblock}
    
    \begin{enumerate}
        \item \textbf{High Cost and Limited Access:} MRI/PET brain scans are expensive and not easily available in primary healthcare clinics.
        \item \textbf{Subjective Cognitive Tests:} Standard written questionnaires (like MMSE) depend on clinician evaluation and take time to conduct.
        \item \textbf{Limitations of Basic Audio Features:} Traditional features like pitch or standard MFCCs alone cannot capture complex speech patterns associated with Alzheimer's.
        \item \textbf{Feature Dimensionality \& Noise:} Combining multi-modal features results in high-dimensional representations requiring robust feature fusion and dimensionality reduction.
    \end{enumerate}
\end{frame}

        \textbf{\Large System Pipeline Overview}
    \end{center}
    \vfill
    \begin{block}{Pipeline Flow (Mid-Term Stage)}
    \begin{itemize}
        \item \textbf{Speech Input:} Spontaneous audio recordings (.wav format) from the ADReSSo 2021 dataset.
        \item \textbf{Audio Preprocessing:} Converting to 16kHz, noise reduction, and removing silent gaps using voice activity detection.
        \item \textbf{Feature Extraction:} Extracting 123 physical acoustic features (MFCCs, Pitch, HNR) + Whisper deep audio embeddings.
        \item \textbf{Feature Fusion \& PCA:} Combining acoustic features and deep embeddings using PCA (reduced to 318 features).
        \item \textbf{Machine Learning Training \& Evaluation:} Training 4 Machine Learning models evaluated via 5-Fold Stratified Cross-Validation.
    \end{itemize}
    \end{block}
\end{frame}

\section{Gaps Identified}
\begin{frame}{Gaps Identified in Current Research}
    \begin{alertblock}{Research Gaps}
        Key issues observed in existing literature that our work addresses:
    \end{alertblock}
    \begin{itemize}
        \item \textbf{Low Accuracy of Simple Acoustic Features:} Official ADReSSo acoustic baselines (eGeMAPS + SVM) achieve only ~65.1\% accuracy.
        \item \textbf{Need for Feature Combination:} Combining hand-crafted acoustic properties with deep audio embeddings improves model sensitivity.
        \item \textbf{High Feature Dimensionality:} Naive feature concatenation introduces redundancy, requiring effective PCA feature fusion.
        \item \textbf{Small Dataset Variance:} Small speech corpora require careful 5-fold cross-validation to prevent data leakage and overfitting.
    \end{itemize}
\end{frame}

% ===== Slide 9: Methodology - Part 1 =====
\section{Methodology}
\begin{frame}{Complete Methodology: Stage 1 -- Audio Preprocessing}
    \begin{block}{Stage 1: Signal Cleaning \& Audio Standardisation}
        Raw speech signals contain ambient room noise, background hums, and long pauses that hurt classifier performance.
    \end{block}
    \begin{itemize}
        \item \textbf{Audio Resampling:} All input \texttt{.wav} recordings are converted to a uniform sample rate of \textbf{16,000 Hz (16 kHz)} mono PCM.
        \item \textbf{Spectral Noise Reduction:} Non-stationary background noise is filtered out using spectral gating via Python's \texttt{noisereduce} library, preserving spoken speech formants.
        \item \textbf{Voice Activity Detection (VAD):} Non-speech silence segments are detected and removed using \texttt{librosa.effects.split(top\_db=25)}, keeping only active speech segments.
        \item \textbf{Data Safety:} Raw audio files remain 100\% untouched; processed audio outputs are exported to \texttt{processed\_audio/}.
    \end{itemize}
\end{frame}

% ===== Slide 10: Methodology - Part 2 =====
\begin{frame}{Complete Methodology: Stage 2 \& 3 -- Feature Extraction \& Fusion}
    \begin{columns}[T]
        \column{0.52\textwidth}
        \textbf{Stage 2: Multimodal Feature Extraction}
        \begin{itemize}
            \item \textbf{123 Acoustic Features:}
                \begin{itemize}
                    \item 40 MFCCs (Mel-Frequency Cepstral Coefficients)
                    \item 40 First-Order Delta ($\Delta$) velocity parameters
                    \item 40 Second-Order Delta-Delta ($\Delta^2$) acceleration
                    \item Pitch ($F_0$) statistics (Mean, Std, Min, Max)
                    \item Harmonics-to-Noise Ratio (HNR) for voice quality
                \end{itemize}
            \item \textbf{Deep Audio Embeddings:}
                \begin{itemize}
                    \item 1024-dimensional deep feature representations extracted from OpenAI's pretrained Whisper encoder.
                \end{itemize}
        \end{itemize}

        \column{0.45\textwidth}
        \textbf{Stage 3: Feature Fusion & PCA}
        \begin{itemize}
            \item \textbf{Vector Concatenation:} Combining 123 acoustic + 1024 deep features $\rightarrow$ 1147-dim representation.
            \item \textbf{Feature Scaling:} Applying \texttt{StandardScaler} (zero mean, unit variance).
            \item \textbf{PCA Reduction:} Performing Principal Component Analysis (PCA) to project the 1147-dim space down to \textbf{318 principal components}, eliminating collinearity while retaining maximum variance.
        \end{itemize}
    \end{columns}
\end{frame}

% ===== Slide 11: Methodology - Part 3 =====
\begin{frame}{Complete Methodology: Stage 4 \& 5 -- ML Classification \& Validation}
    \begin{columns}[T]
        \column{0.52\textwidth}
        \textbf{Stage 4: Machine Learning Classifiers}
        \begin{itemize}
            \item \textbf{Logistic Regression:} Linear classifier with L2 regularization for robust probability estimation.
            \item \textbf{SVM (RBF Kernel):} Non-linear Support Vector Machine mapping to high-dimensional margin space ($C=1.0$).
            \item \textbf{Random Forest:} Ensemble classifier with 100 decision trees.
            \item \textbf{K-Nearest Neighbors (KNN):} Non-parametric instance-based classifier ($k=5$).
        \end{itemize}

        \column{0.45\textwidth}
        \textbf{Stage 5: Evaluation Protocol}
        \begin{itemize}
            \item \textbf{5-Fold Stratified Cross-Validation:}
                \begin{itemize}
                    \item Dataset split into 5 equal folds, preserving class balance (87 AD / 79 Healthy).
                    \item 4 folds (80\%) for training, 1 fold (20\%) for validation per iteration.
                \end{itemize}
            \item \textbf{Evaluation Metrics:}
                \begin{itemize}
                    \item Accuracy (\%) $\pm$ Standard Deviation
                    \item Precision, Recall, F1-Score
                    \item Receiver Operating Characteristic AUC (ROC-AUC)
                \end{itemize}
        \end{itemize}
    \end{columns}
\end{frame}

% ===== Slide 16: Performance Analysis & Key Findings =====
\begin{frame}{Performance Analysis \& Key Findings}
    \begin{block}{Experimental Observations (Machine Learning Stage)}
    \begin{itemize}
        \item \textbf{Strong Class Separation:} Logistic Regression achieved the highest discriminatory capability with an ROC-AUC of \textbf{0.7815} and an F1-Score of \textbf{0.7117}.
        \item \textbf{Support Vector Performance:} SVM with RBF Kernel achieved \textbf{66.29\% accuracy} and high recall (0.7346), capturing non-linear boundary characteristics.
        \item \textbf{Superiority Over Acoustic Baselines:} Combining 123 physical acoustic features with Whisper deep embeddings yielded a \textbf{+5.4\% accuracy gain} over standard eGeMAPS acoustic features (65.1\%).
        \item \textbf{Robust Validation:} 5-Fold Stratified Cross-Validation confirmed model stability and prevented data leakage across folds.
    \end{itemize}
    \end{block}
\end{frame}
