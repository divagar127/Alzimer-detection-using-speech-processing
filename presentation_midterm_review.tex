% ===== Slide 3: Introduction and Motivation =====
\section{Introduction}
\begin{frame}{Introduction and Motivation}
    \begin{block}{Clinical Context}
        Alzheimer's Disease (AD) is a progressive neurodegenerative disorder leading to irreversible memory loss, linguistic degradation, and cognitive impairment.
    \end{block}
    \begin{itemize}
        \item \textbf{Motivation:} Conventional clinical diagnostics depend on expensive neuroimaging (Brain MRI/PET) or invasive lumbar punctures. Spontaneous speech analysis offers a \textbf{non-invasive, rapid, and low-cost} screening alternative.
        \item \textbf{Acoustic Biomarkers:} Early AD manifests through subtle speech impairments—pitch instability, micro-pauses, dysfluency, reduced harmonicity, and articulation slowing.
        \item \textbf{Explainable AI (XAI) Imperative:} Integrating transparent machine learning ensures clinical trust and interpretable diagnostic reasoning for clinicians.
    \end{itemize}
\end{frame}

% ===== Slide 4: Problem Definition =====
\section{Problem Definition}
\begin{frame}{Problem Definition}
    \begin{alertblock}{The Core Research Challenge}
        Developing an automated, non-invasive, and interpretable AD detection system that overcomes diagnostic bottlenecks and surpasses traditional acoustic baselines.
    \end{alertblock}
    
    \begin{enumerate}
        \item \textbf{Clinical Diagnostic Bottlenecks:} Neuroimaging (MRI/PET) is costly and inaccessible in rural healthcare. Cognitive screeners (MMSE, MoCA) suffer from subjective scorer bias and low sensitivity to early-stage onset.
        \item \textbf{Limitations of Traditional Speech Feature Sets:} Standard hand-engineered acoustic features (MFCCs, jitter, shimmer) alone fail to capture multi-scale temporal dependencies and deep semantic representations of AD speech.
        \item \textbf{Lack of Interpretability:} Black-box deep networks do not explain \textit{why} a subject is flagged, hindering clinical adoption.
    \end{enumerate}
\end{frame}

% ===== Slide 5: Project Overview =====
\begin{frame}{Project Architecture Flow (40--50\% Scope)}
    \begin{center}
        \textbf{\Large Pipeline Architecture Flow}
    \end{center}
    \vfill
    \begin{block}{Implemented Processing Pipeline}
    \begin{itemize}
        \item \textbf{Audio Input:} Spontaneous speech recordings from ADReSSo Challenge dataset (.wav format).
        \item \textbf{Stage 1 (Preprocessing):} 16kHz resampling, spectral gating noise reduction, VAD silence removal.
        \item \textbf{Stage 2 (Feature Extraction):} 123-dim hand-crafted acoustic features + 1024-dim Whisper deep embeddings.
        \item \textbf{Stage 3 (Feature Fusion):} Multimodal feature concatenation + PCA reduction to 318 dimensions.
        \item \textbf{Stage 4 \& 5 (Classification \& Evaluation):} ML (SVM, RF, LR) + DL (PyTorch 3-Layer DNN, 1D-CNN) evaluated via \textbf{5-Fold Stratified Cross-Validation}.
    \end{itemize}
    \end{block}
\end{frame}

% ===== Slide 6: Gaps Identified =====
\section{Gaps Identified}
\begin{frame}{Gaps Identified in Existing Literature}
    \begin{alertblock}{Key Research Gaps}
        Critical limitations present in existing computational healthcare literature:
    \end{alertblock}
    \begin{itemize}
        \item \textbf{Underutilization of Deep Multimodal Fusion:} Prior studies rely either purely on hand-crafted acoustic parameters or uninterpretable text transcripts, missing cross-modal interactions.
        \item \textbf{Black-Box Clinical Barrier:} Existing high-performing models lack feature attribution (SHAP/LIME), making predictions opaque to medical practitioners.
        \item \textbf{Acoustic Baseline Underperformance:} The official ADReSSo acoustic baseline (eGeMAPS + SVM) achieves only \textbf{65.1\% accuracy}, highlighting the need for advanced deep embedding integration.
        \item \textbf{Evaluation Rigor:} High variance in small speech corpora requires strict \textbf{5-Fold Stratified Cross-Validation} to prevent data leakage.
    \end{itemize}
\end{frame}

% ===== Slide 7: Objectives =====
\section{Objectives}
\begin{frame}{Objectives \& Technical Scope}
    \begin{block}{Main Research Objective}
        To design, implement, and validate an automated speech processing pipeline for Alzheimer's Detection that outperforms official challenge baselines while establishing a foundation for Explainable AI.
    \end{block}
    
    \textbf{Specific Milestone Objectives (Completed in 40--50\% Phase):}
    \begin{enumerate}
        \item Build a standardized audio preprocessing pipeline (spectral denoising, 16kHz resampling, VAD segmentation).
        \item Extract a 123-dimensional acoustic feature space (40 MFCCs, Deltas, Pitch F0 statistics, HNR) combined with deep Whisper embeddings.
        \item Implement PCA-based multimodal feature fusion (318-dim representation space).
        \item Train and systematically evaluate ML (Logistic Regression, SVM RBF, Random Forest) and DL (PyTorch DNN, 1D-CNN) models using 5-Fold Stratified CV.
    \end{enumerate}
\end{frame}

% ===== Slide 8: Methodology =====
\section{Methodology}
\begin{frame}{Detailed Methodology \& Technical Workflow}
    \begin{columns}[T]
        \column{0.52\textwidth}
        \textbf{5-Stage Implementation Workflow:}
        \begin{enumerate}
            \item \textbf{Data Preprocessing:} Resampling to 16kHz, spectral noise reduction via \texttt{noisereduce}, voice activity detection (\texttt{librosa.effects.split}, top\_db=25).
            \item \textbf{Feature Extraction:} 
                \begin{itemize}
                    \item Acoustic: 40 MFCCs + $\Delta$ + $\Delta^2$, Pitch F0 (Mean, Std), Harmonics-to-Noise Ratio (HNR).
                    \item Deep: 1024-dim Whisper pretrained representations.
                \end{itemize}
            \item \textbf{Feature Fusion:} Concatenation (1147-dim) $\rightarrow$ StandardScaler $\rightarrow$ PCA (318 dimensions).
            \item \textbf{Classification Models:} SVM (RBF), Logistic Regression, Random Forest, PyTorch 3-Layer DNN, PyTorch 1D-CNN.
            \item \textbf{Evaluation:} 5-Fold Stratified Cross-Validation.
        \end{enumerate}
        
        \column{0.45\textwidth}
        \textbf{System Block Diagram:}
        \begin{center}
            \fbox{\parbox{0.95\textwidth}{\centering
                \small
                \textbf{Raw Speech Audio (.wav)} \\

                \textbf{16kHz Denoising \& VAD} \\
                $\downarrow$ \\
                \textbf{Acoustic (123d) + Whisper (1024d)} \\
                $\downarrow$ \\
                \textbf{StandardScaler + PCA (318d)} \\
                $\downarrow$ \\
                \textbf{ML \& PyTorch DL Classifiers} \\
                $\downarrow$ \\
                \textbf{5-Fold Stratified CV Output}

        \end{center}

\end{frame}
